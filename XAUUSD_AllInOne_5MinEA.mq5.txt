// Rename to XAUUSD_AllInOne_5MinEA.mq5 before compiling.
// One-chart EA: visual 5-minute plan + optional auto trading + mandatory SL/TP.
#property strict
#include <Trade/Trade.mqh>

input string InpSymbol="";
input bool InpAutoTrade=false;                 // Keep false until tested on demo
input double InpLot=0.01;
input double InpMaxRiskMoney=1.40;
input long InpMagic=20260824;
input double InpStopATR=1.80;
input double InpRewardRisk=1.50;
input int InpMaxSpreadPoints=80;
input int InpEntryOffsetPoints=2;
input bool InpPopupAlert=false;

CTrade trade;
string tradeSymbol,previousSignal="",prefix="AllInOne_";
int m1Fast=INVALID_HANDLE,m1Slow=INVALID_HANDLE,rsi=INVALID_HANDLE,atr=INVALID_HANDLE,m5Fast=INVALID_HANDLE,m5Slow=INVALID_HANDLE;
datetime lastM1Bar=0;

int OnInit()
{
   tradeSymbol=(InpSymbol=="" ? _Symbol : InpSymbol);
   if(!SymbolSelect(tradeSymbol,true)) return INIT_FAILED;
   m1Fast=iMA(tradeSymbol,PERIOD_M1,9,0,MODE_EMA,PRICE_CLOSE);
   m1Slow=iMA(tradeSymbol,PERIOD_M1,21,0,MODE_EMA,PRICE_CLOSE);
   rsi=iRSI(tradeSymbol,PERIOD_M1,14,PRICE_CLOSE);
   atr=iATR(tradeSymbol,PERIOD_M1,14);
   m5Fast=iMA(tradeSymbol,PERIOD_M5,20,0,MODE_EMA,PRICE_CLOSE);
   m5Slow=iMA(tradeSymbol,PERIOD_M5,50,0,MODE_EMA,PRICE_CLOSE);
   if(m1Fast==INVALID_HANDLE || m1Slow==INVALID_HANDLE || rsi==INVALID_HANDLE || atr==INVALID_HANDLE || m5Fast==INVALID_HANDLE || m5Slow==INVALID_HANDLE) return INIT_FAILED;
   trade.SetExpertMagicNumber(InpMagic); trade.SetDeviationInPoints(20);
   Comment("XAUUSD All-in-One\nLoading data...");
   return INIT_SUCCEEDED;
}

void OnDeinit(const int reason)
{
   IndicatorRelease(m1Fast); IndicatorRelease(m1Slow); IndicatorRelease(rsi); IndicatorRelease(atr); IndicatorRelease(m5Fast); IndicatorRelease(m5Slow);
   ClearLevels(); Comment("");
}

void OnTick()
{
   EnsureSLTP();
   datetime current=iTime(tradeSymbol,PERIOD_M1,0);
   if(current==0 || current==lastM1Bar) return;
   lastM1Bar=current;
   AnalyzeAndAct();
}

void AnalyzeAndAct()
{
   MqlTick tick; double point=SymbolInfoDouble(tradeSymbol,SYMBOL_POINT);
   if(!SymbolInfoTick(tradeSymbol,tick) || point<=0) return;
   double spread=(tick.ask-tick.bid)/point;
   if(spread>InpMaxSpreadPoints) { ShowWait("WAIT — spread is too high: "+DoubleToString(spread,0)+" points."); return; }
   double f[],s[],rr[],aa[],hf[],hs[]; MqlRates c[];
   ArraySetAsSeries(f,true); ArraySetAsSeries(s,true); ArraySetAsSeries(rr,true); ArraySetAsSeries(aa,true); ArraySetAsSeries(hf,true); ArraySetAsSeries(hs,true); ArraySetAsSeries(c,true);
   if(CopyBuffer(m1Fast,0,0,3,f)<3 || CopyBuffer(m1Slow,0,0,3,s)<3 || CopyBuffer(rsi,0,0,3,rr)<3 || CopyBuffer(atr,0,0,3,aa)<3 || CopyBuffer(m5Fast,0,0,2,hf)<2 || CopyBuffer(m5Slow,0,0,2,hs)<2 || CopyRates(tradeSymbol,PERIOD_M1,0,3,c)<3) return;
   bool buy=hf[1]>hs[1] && f[1]>s[1] && c[1].close>f[1] && rr[1]>=52 && rr[1]<=70;
   bool sell=hf[1]<hs[1] && f[1]<s[1] && c[1].close<f[1] && rr[1]>=30 && rr[1]<=48;
   if(!buy && !sell)
   {
      string trend=(hf[1]>hs[1] ? "UP" : (hf[1]<hs[1] ? "DOWN" : "NEUTRAL"));
      ShowWait("WAIT — M5 trend: "+trend+" | RSI: "+DoubleToString(rr[1],1)+". No confirmed setup.");
      return;
   }
   bool isBuy=buy;
   double distance=SafeDistance(aa[1]);
   if(distance<=0) { ShowWait("WAIT — stop distance needed for 0.01 lot exceeds the risk limit."); return; }
   double entry=(isBuy ? c[1].high+InpEntryOffsetPoints*point : c[1].low-InpEntryOffsetPoints*point);
   double sl=(isBuy ? entry-distance : entry+distance);
   double tp=(isBuy ? entry+distance*InpRewardRisk : entry-distance*InpRewardRisk);
   int digits=(int)SymbolInfoInteger(tradeSymbol,SYMBOL_DIGITS);
   string side=(isBuy ? "BUY" : "SELL");
   DrawLevel("ENTRY",entry,isBuy ? clrLime : clrTomato); DrawLevel("SL",sl,clrRed); DrawLevel("TP",tp,clrDeepSkyBlue);
   string plan="XAUUSD 5-MINUTE PLAN\nAction: "+side+"\nEntry: "+DoubleToString(entry,digits)+"\nSL: "+DoubleToString(sl,digits)+"\nTP: "+DoubleToString(tp,digits)+"\nValid until: "+TimeToString(TimeCurrent()+300,TIME_MINUTES)+"\nAuto Trade: "+(InpAutoTrade ? "ON" : "OFF");
   Comment(plan);
   string signal=side+DoubleToString(entry,digits);
   if(InpPopupAlert && previousSignal!=signal) Alert(plan);
   previousSignal=signal;
   if(InpAutoTrade && !PositionSelect(tradeSymbol)) OpenWithSLTP(isBuy,entry,sl,tp);
}

double SafeDistance(double atrValue)
{
   double tickSize=SymbolInfoDouble(tradeSymbol,SYMBOL_TRADE_TICK_SIZE),tickValue=SymbolInfoDouble(tradeSymbol,SYMBOL_TRADE_TICK_VALUE),point=SymbolInfoDouble(tradeSymbol,SYMBOL_POINT);
   if(tickSize<=0 || tickValue<=0 || point<=0 || InpLot<=0) return 0;
   double maxDistance=InpMaxRiskMoney/(tickValue*InpLot)*tickSize;
   double minDistance=(int)SymbolInfoInteger(tradeSymbol,SYMBOL_TRADE_STOPS_LEVEL)*point+point;
   double wanted=MathMin(atrValue*InpStopATR,maxDistance);
   return (wanted>=minDistance ? wanted : 0);
}

void OpenWithSLTP(bool buy,double entry,double sl,double tp)
{
   int digits=(int)SymbolInfoInteger(tradeSymbol,SYMBOL_DIGITS);
   bool sent=(buy ? trade.Buy(InpLot,tradeSymbol,0,NormalizeDouble(sl,digits),NormalizeDouble(tp,digits),"All-in-One") : trade.Sell(InpLot,tradeSymbol,0,NormalizeDouble(sl,digits),NormalizeDouble(tp,digits),"All-in-One"));
   if(!sent) Print("Order rejected: ",trade.ResultRetcodeDescription());
   EnsureSLTP();
}

void EnsureSLTP()
{
   if(!PositionSelect(tradeSymbol) || (long)PositionGetInteger(POSITION_MAGIC)!=InpMagic) return;
   if(PositionGetDouble(POSITION_SL)>0 && PositionGetDouble(POSITION_TP)>0) return;
   double aa[]; ArraySetAsSeries(aa,true); if(CopyBuffer(atr,0,0,1,aa)!=1) return;
   double distance=SafeDistance(aa[0]); if(distance<=0) return;
   ENUM_POSITION_TYPE type=(ENUM_POSITION_TYPE)PositionGetInteger(POSITION_TYPE);
   double open=PositionGetDouble(POSITION_PRICE_OPEN),sl=PositionGetDouble(POSITION_SL),tp=PositionGetDouble(POSITION_TP);
   if(type==POSITION_TYPE_BUY) { if(sl==0) sl=open-distance; if(tp==0) tp=open+distance*InpRewardRisk; }
   else { if(sl==0) sl=open+distance; if(tp==0) tp=open-distance*InpRewardRisk; }
   int digits=(int)SymbolInfoInteger(tradeSymbol,SYMBOL_DIGITS);
   if(!trade.PositionModify(tradeSymbol,NormalizeDouble(sl,digits),NormalizeDouble(tp,digits))) Print("SL/TP retry failed: ",trade.ResultRetcodeDescription());
}

void ShowWait(string reason)
{
   ClearLevels(); previousSignal="WAIT";
   Comment("XAUUSD 5-MINUTE PLAN\n"+reason+"\nNo trade suggested for the next 5 minutes.\nAuto Trade: "+(InpAutoTrade ? "ON" : "OFF"));
}

void DrawLevel(string label,double value,color clr)
{
   string name=prefix+label;
   if(ObjectFind(0,name)<0) ObjectCreate(0,name,OBJ_HLINE,0,0,value);
   ObjectSetDouble(0,name,OBJPROP_PRICE,value); ObjectSetInteger(0,name,OBJPROP_COLOR,clr); ObjectSetInteger(0,name,OBJPROP_WIDTH,2);
}

void ClearLevels()
{
   ObjectDelete(0,prefix+"ENTRY"); ObjectDelete(0,prefix+"SL"); ObjectDelete(0,prefix+"TP");
}
